---
title: <ScriptGoogleMapsPinElement>
---

可自定义的图钉标记。作为 [`<ScriptGoogleMapsAdvancedMarkerElement>`{lang="html"}](/scripts/google-maps/api/advanced-marker-element) 的子元素放置。

::script-types{script-key="google-maps" filter="ScriptGoogleMapsPinElement"}
::

## 用法

```vue
<template>
  <ScriptGoogleMaps api-key="你的-api-密钥">
    <ScriptGoogleMapsAdvancedMarkerElement
      :position="{ lat: -34.407, lng: 150.654 }"
    >
      <ScriptGoogleMapsPinElement
        :options="{
          scale: 1.5,
          background: '#FF0000',
          borderColor: '#CC0000',
          glyphColor: '#FFFFFF',
        }"
      />
    </ScriptGoogleMapsAdvancedMarkerElement>
  </ScriptGoogleMaps>
</template>
```

### 文本图标

在图钉内显示字符或短标签。

```vue
<template>
  <ScriptGoogleMaps api-key="你的-api-密钥">
    <ScriptGoogleMapsAdvancedMarkerElement
      :position="{ lat: -34.397, lng: 150.644 }"
    >
      <ScriptGoogleMapsPinElement
        :options="{
          glyph: 'A',
          glyphColor: '#FFFFFF',
          background: '#4285F4',
          borderColor: '#1a73e8',
        }"
      />
    </ScriptGoogleMapsAdvancedMarkerElement>
  </ScriptGoogleMaps>
</template>
```

### 隐藏标记

将 `glyph` 设置为空字符串，以获得纯色图钉，不带任何图标或文字。

```vue
<template>
  <ScriptGoogleMaps api-key="你的-api-密钥">
    <ScriptGoogleMapsAdvancedMarkerElement
      :position="{ lat: -34.397, lng: 150.644 }"
    >
      <ScriptGoogleMapsPinElement
        :options="{
          glyph: '',
          background: '#FBBC04',
          scale: 1.2,
        }"
      />
    </ScriptGoogleMapsAdvancedMarkerElement>
  </ScriptGoogleMaps>
</template>
```

### 多种图钉样式

组合不同的图钉配置来区分标记类别。

```vue
<script setup lang="ts">
const restaurants = [{ lat: -34.397, lng: 150.644 }]
const hotels = [{ lat: -34.407, lng: 150.654 }]

const restaurantPin = {
  background: '#EA4335',
  glyphColor: '#fff',
  glyph: '🍕',
}

const hotelPin = {
  background: '#4285F4',
  glyphColor: '#fff',
  glyph: '🏨',
}
</script>

<template>
  <ScriptGoogleMaps api-key="你的-api-密钥">
    <ScriptGoogleMapsAdvancedMarkerElement
      v-for="(pos, i) in restaurants"
      :key="`r-${i}`"
      :position="pos"
    >
      <ScriptGoogleMapsPinElement :options="restaurantPin" />
    </ScriptGoogleMapsAdvancedMarkerElement>

    <ScriptGoogleMapsAdvancedMarkerElement
      v-for="(pos, i) in hotels"
      :key="`h-${i}`"
      :position="pos"
    >
      <ScriptGoogleMapsPinElement :options="hotelPin" />
    </ScriptGoogleMapsAdvancedMarkerElement>
  </ScriptGoogleMaps>
</template>
```

::callout
`options` 属性是响应式的。更改它会就地更新图钉外观。如果你需要完全自定义的 HTML 而不是图钉形状，请改用 `AdvancedMarkerElement` 上的 `#content` 插槽。
::
